# ATC 自动换刀功能 — 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 grblHAL 添加刀架式自动换刀（ATC）功能，M6 触发同步运动序列完成还刀-取刀流程。

**Architecture:** 新增 `atc_plugin.c/h` 作为 ATC 插件，在 `my_plugin_init()` 中调用 `atc_init()` 注册 HAL 回调（`hal.tool.change`、`hal.tool.select`、`hal.tool.atc_get_state`），设置 `hal.driver_cap.atc=1`。换刀序列使用 `mc_line()` + `protocol_buffer_synchronize()` 同步执行。ATC 设置（$900-$907）通过 `settings_register()` 注册。

**Tech Stack:** C (C11), bare-metal ARM Cortex-M4F, grblHAL core API, AT32F407 standard peripheral library

```c
/**
 * @file   atc_plugin.h
 * @brief  ATC (Automatic Tool Changer) plugin for rack-style tool changer
 *
 * Hardware: single-row linear rack, 5-8 pockets, pneumatic drawbar
 * I/O:      PB0 = drawbar solenoid (HIGH = release)
 *           PB1 = air blast (HIGH = blast)
 */
#ifndef __ATC_PLUGIN_H__
#define __ATC_PLUGIN_H__

#include "grbl.h"
#include "hal.h"
#include "core_handlers.h"

/* ---- ATC settings struct ---- */
typedef union {
    uint8_t value;
    struct {
        uint8_t air_blast    :1,  /* bit0: enable air blast during tool change */
                restore_xy   :1,  /* bit1: restore XY position after M6 */
                unassigned   :6;
    };
} atc_flags_t;

typedef struct {
    uint8_t  n_pockets;       /* $900 - number of rack pockets (1-16)        */
    float    pocket_pitch;    /* $901 - X-axis spacing between pockets (mm)  */
    float    rack_origin_x;   /* $902 - X coordinate of pocket #1 (machine)  */
    float    rack_origin_y;   /* $903 - Y coordinate of rack (machine)       */
    float    z_clear_height;  /* $904 - Z safe height above rack (machine)   */
    float    z_pickup_depth;  /* $905 - Z depth for pickup/drop (machine)    */
    uint16_t draw_delay_ms;   /* $906 - drawbar actuation delay (ms)         */
    atc_flags_t flags;        /* $907 - option bits                          */
} atc_settings_t;

/* ---- Default values ---- */
#define ATC_DEFAULT_POCKETS        8
#define ATC_DEFAULT_POCKET_PITCH   50.0f
#define ATC_DEFAULT_RACK_ORIGIN_X  0.0f
#define ATC_DEFAULT_RACK_ORIGIN_Y  0.0f
#define ATC_DEFAULT_Z_CLEAR        0.0f
#define ATC_DEFAULT_Z_PICKUP       0.0f
#define ATC_DEFAULT_DRAW_DELAY_MS  500

/* ---- Plugin entry point ---- */
void atc_init(void);

#endif /* __ATC_PLUGIN_H__ */
```

```c
static void atc_gpio_init(void)
{
    gpio_init_type gpio_init_struct;
    gpio_default_para_init(&gpio_init_struct);
    gpio_init_struct.gpio_mode       = GPIO_MODE_OUTPUT;
    gpio_init_struct.gpio_out_type   = GPIO_OUTPUT_PUSH_PULL;
    gpio_init_struct.gpio_drive_strength = GPIO_DRIVE_STRENGTH_STRONGER;

    /* PB0 — drawbar solenoid */
    gpio_init_struct.gpio_pins = ATC_DRAW_BAR_PIN;
    gpio_init(ATC_DRAW_BAR_PORT, &gpio_init_struct);
    gpio_bits_reset(ATC_DRAW_BAR_PORT, ATC_DRAW_BAR_PIN);  /* idle = clamped */

    /* PB1 — air blast */
    gpio_init_struct.gpio_pins = ATC_AIR_BLAST_PIN;
    gpio_init(ATC_AIR_BLAST_PORT, &gpio_init_struct);
    gpio_bits_reset(ATC_AIR_BLAST_PORT, ATC_AIR_BLAST_PIN);  /* idle = off */
}

static void atc_drawbar(bool release)
{
    if (release) {
        gpio_bits_set(ATC_DRAW_BAR_PORT, ATC_DRAW_BAR_PIN);
    } else {
        gpio_bits_reset(ATC_DRAW_BAR_PORT, ATC_DRAW_BAR_PIN);
    }
}

static void atc_air_blast(bool on)
{
    if (on) {
        gpio_bits_set(ATC_AIR_BLAST_PORT, ATC_AIR_BLAST_PIN);
    } else {
        gpio_bits_reset(ATC_AIR_BLAST_PORT, ATC_AIR_BLAST_PIN);
    }
}
```

```c
/* ---- Compute pocket position in machine coordinates ---- */
static void atc_get_pocket_pos(uint8_t pocket, float *x, float *y)
{
    *x = s_atc.rack_origin_x + (float)(pocket - 1) * s_atc.pocket_pitch;
    *y = s_atc.rack_origin_y;
}

/* ---- Move Z to target height, rapid, sync ---- */
static bool atc_move_z(float z_target, bool rapid)
{
    plan_line_data_t plan_data;
    float target[N_AXIS];

    system_convert_array_steps_to_mpos(target, sys.position);
    plan_data_init(&plan_data);
    plan_data.condition.rapid_motion = rapid ? On : Off;

    /* Keep XY as is, change Z only */
    target[Z_AXIS] = z_target;

    return mc_line(target, &plan_data) && protocol_buffer_synchronize();
}

/* ---- Move XY to position, rapid, sync ---- */
static bool atc_move_xy(float x, float y, float z_keep)
{
    plan_line_data_t plan_data;
    float target[N_AXIS];

    system_convert_array_steps_to_mpos(target, sys.position);
    plan_data_init(&plan_data);
    plan_data.condition.rapid_motion = On;

    target[X_AXIS] = x;
    target[Y_AXIS] = y;
    target[Z_AXIS] = z_keep;

    return mc_line(target, &plan_data) && protocol_buffer_synchronize();
}
```

```c
/* ---- Execute full tool change sequence ---- */
static status_code_t atc_tool_change(parser_state_t *parser_state)
{
    /* Parameter validation */
    if (s_pending_tool == NULL)
        return Status_GCodeToolError;

    tool_id_t old_id = parser_state->tool->tool_id;
    tool_id_t new_id = s_pending_tool->tool_id;

    /* Same tool — no-op */
    if (old_id == new_id)
        return Status_OK;

    /* Validate pocket range */
    if (old_id > s_atc.n_pockets || new_id > s_atc.n_pockets)
        return Status_GCodeToolError;

    plan_line_data_t plan_data;
    float target[N_AXIS];
    float saved_xy[2];
    bool ok;

    plan_data_init(&plan_data);

    /* 1. Stop spindle and coolant */
    spindle_all_off(false);
    hal.coolant.set_state((coolant_state_t){0});

    /* 2. Save current XY position (before applying tool offset) */
    system_convert_array_steps_to_mpos(target, sys.position);
    /* Strip tool offset to get spindle nose machine position */
    target[Z_AXIS] -= gc_get_offset(Z_AXIS, false);
    saved_xy[0] = target[X_AXIS];
    saved_xy[1] = target[Y_AXIS];

    /* 3. Return old tool (if spindle has one) */
    if (old_id > 0) {
        float px, py;
        atc_get_pocket_pos((uint8_t)old_id, &px, &py);

        /* 3a. Z to safe height */
        ok = atc_move_z(s_atc.z_clear_height, true);
        if (!ok) goto fail;

        /* 3b. XY to old pocket */
        ok = atc_move_xy(px, py, s_atc.z_clear_height);
        if (!ok) goto fail;

        /* 3c. Z down to pickup depth */
        ok = atc_move_z(s_atc.z_pickup_depth, true);
        if (!ok) goto fail;

        /* 3d. Release drawbar + optional air blast */
        atc_drawbar(true);
        if (s_atc.flags.air_blast)
            atc_air_blast(true);
        hal.delay_ms(s_atc.draw_delay_ms, NULL);
        atc_air_blast(false);

        /* 3e. Z up to safe height (tool stays in pocket) */
        ok = atc_move_z(s_atc.z_clear_height, true);
        atc_drawbar(false);
        if (!ok) goto fail;
    }

    /* 4. Pick up new tool (if not T0) */
    if (new_id > 0) {
        float px, py;
        atc_get_pocket_pos((uint8_t)new_id, &px, &py);

        /* 4a. XY to new pocket above */
        ok = atc_move_xy(px, py, s_atc.z_clear_height);
        if (!ok) goto fail;

        /* 4b. Z down to pickup depth */
        ok = atc_move_z(s_atc.z_pickup_depth, true);
        if (!ok) goto fail;

        /* 4c. Optional air blast, clamp drawbar */
        if (s_atc.flags.air_blast)
            atc_air_blast(true);
        atc_drawbar(false);
        hal.delay_ms(s_atc.draw_delay_ms, NULL);
        atc_air_blast(false);

        /* 4d. Z up to safe height with tool */
        ok = atc_move_z(s_atc.z_clear_height, true);
        if (!ok) goto fail;
    }

    /* 5. Optional: restore XY */
    if (s_atc.flags.restore_xy) {
        plan_data_init(&plan_data);
        plan_data.condition.rapid_motion = On;

        system_convert_array_steps_to_mpos(target, sys.position);
        target[X_AXIS] = saved_xy[0];
        target[Y_AXIS] = saved_xy[1];

        ok = mc_line(target, &plan_data) && protocol_buffer_synchronize();
        if (!ok) goto fail;
    }

    /* 6. Sync final position */
    sync_position();

    /* 7. Notify core of success */
    s_pending_tool = NULL;  /* consumed */
    return Status_OK;

fail:
    s_pending_tool = NULL;
    return Status_Reset;
}
```

```c
/* ---- Plugin entry: called from my_plugin_init() ---- */
void atc_init(void)
{
    /* 1. Init GPIO */
    atc_gpio_init();

    /* 2. Register settings */
    settings_register(&atc_setting_details);

    /* 3. Register HAL callbacks */
    hal.tool.select       = atc_tool_select;
    hal.tool.change       = atc_tool_change;
    hal.tool.atc_get_state = atc_get_state;

    /* Set driver capability flag — this tells the core:
       - ATC hardware exists (gcode.c checks this for M6 path)
       - Skip tc_init() (grbllib.c checks this) */
    hal.driver_cap.atc = 1;

    MLOGF(I, "ATC: Rack ATC plugin initialized, %u pockets\r\n",
          (unsigned int)s_atc.n_pockets);
}
```
